        page: currentPage,
        limit: itemsPerPage,

        queryParams.status = selectedStatus === "active" ? "true" : "false";

        queryParams.category = selectedCategory;

        queryParams.brand = selectedBrand;
      }

        queryParams.collection = selectedCollection;

        queryParams.search = searchQuery;

        queryParams.startDate = dateRange.startDate;
        queryParams.endDate = dateRange.endDate;

        setTotalItems(response.pagination.totalItems);
        setTotalPages(response.pagination.totalPages);
        setFilteredCount(response.pagination.filteredCount);

        toast.error(response.message || "Failed to fetch products");
      }

      console.error("Error fetching paginated products:", error);
      toast.error("Failed to fetch products");
    } finally {
      setPaginatedLoading(false);

                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          <div className="flex gap-2 justify-center items-center">
                            {product?.img_path ? (
                              <img
                                src={product?.img_path}
                                alt={product?.name}
                                className="h-14 object-contain rounded-lg"
                              />
                            ) : (
                              ""
                            )}
                          </div>
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          <UniqueCode
                            text={product?.brand?.code}
                            codeName="Brand"
                          />
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.brand?.name}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          <UniqueCode
                            text={product?.subBrand?.code}
                            codeName="Sub Brand"
                          />
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.subBrand?.name}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          <UniqueCode
                            text={product?.cat_id?.code}
                            codeName="Category"
                          />
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.cat_id?.name}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          <UniqueCode
                            text={product?.collection_id?.code}
                            codeName="Collection"
                          />
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.collection_id?.name}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          <UniqueCode
                            text={product?.sku_group_id}
                            codeName="SKU Group"
                          />
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.sku_group__name}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.size}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.product_type}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          <UniqueCode
                            text={product?.product_hsn_code}
                            codeName="HSN"
                          />
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.no_of_pieces_in_a_box}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.cgst}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.sgst}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.igst}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.sbu}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.base_point}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.uom}
                        </Table.Cell>
                        <Table.Cell className="text-gray-900 dark:text-gray-200 whitespace-nowrap">
                          {product?.createdAt
                            ? moment(product?.createdAt)
                                .tz("Asia/Kolkata")
                                .format("YYYY-MM-DD")
                            : ""}
                        </Table.Cell>
                        <Table.Cell>
                          <StatusIndicator
                            status={product.status}
                            onClick={() => handleStatusUpdate(product)}
                          />
                        </Table.Cell>
                        <Table.Cell>
                          <div className="flex gap-2 justify-center items-center">
                            <EditButton
                              onClick={() => handleSetEdit(product)}
                            />
                          </div>
                        </Table.Cell>
                      </Table.Row>
                    ))
                  ) : (
                    <Table.Row>
                      <Table.Cell colSpan={"100%"} className="text-center">
                        No products found
                      </Table.Cell>
                    </Table.Row>
                  )}
                </Table.Body>
              </Table>
            </div>
          )}
        </div>
      </div>

      {/* Add/Edit Product Modal */}
      <Modal show={openModal} onClose={onCloseModal}>
        <Modal.Header>
          {modalMode === "add" ? "Add Product" : "Edit Product"}
        </Modal.Header>
        <Modal.Body>
          {/* Form fields - implementation abbreviated */}
        </Modal.Body>
        <Modal.Footer>
          {modalMode === "add" ? (
            <Button onClick={handleAddProduct} disabled={formLoading}>
              {formLoading ? (
                <Spinner size="sm" aria-label="Loading spinner" />
              ) : (
                "Add Product"
              )}
            </Button>
          ) : (
            <Button onClick={handleEditProduct} disabled={formLoading}>
              {formLoading ? (
                <Spinner size="sm" aria-label="Loading spinner" />
              ) : (
                "Update Product"
              )}
            </Button>
          )}
        </Modal.Footer>
      </Modal>
    </>
  );
};

export default Product;
